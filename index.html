Price history
Pick a symbol to chart its recorded price history.

Historical price charts need one more table in your Supabase project. This is optional — everything else works without it.

Show setup SQL 
create table if not exists price_history (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null default auth.uid() references auth.users(id),
  symbol text not null,
  price numeric not null,
  recorded_at timestamptz not null default now()
);
alter table price_history enable row level security;
create policy "Users manage their own price history"
  on price_history for all
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);
create index if not exists price_history_user_symbol_idx
  on price_history (user_id, symbol, recorded_at);
